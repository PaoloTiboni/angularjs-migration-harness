/*
 * <pratiche-list pratiche="..." on-seleziona="...">
 * Lista con filtro testuale e per stato. Usa il template esterno (templateUrl).
 */
(function () {
  'use strict';

  angular.module('pratiche').component('praticheList', {
    templateUrl: 'components/pratiche-list/pratiche-list.html',
    bindings: {
      pratiche: '<',
      onSeleziona: '&'
    },
    controller: function () {
      var ctrl = this;

      ctrl.testo = '';
      ctrl.stato = '';
      ctrl.visibili = [];

      function filtra() {
        var testo = ctrl.testo.trim().toLowerCase();
        ctrl.visibili = (ctrl.pratiche || []).filter(function (p) {
          if (ctrl.stato && p.stato !== ctrl.stato) {
            return false;
          }
          if (testo && (p.oggetto + ' ' + p.richiedente).toLowerCase().indexOf(testo) === -1) {
            return false;
          }
          return true;
        });
      }

      ctrl.$onChanges = function (changes) {
        if (changes.pratiche) {
          filtra();
        }
      };

      ctrl.cambiaFiltro = function () {
        filtra();
      };

      ctrl.seleziona = function (pratica) {
        ctrl.onSeleziona({ pratica: pratica });
      };

      ctrl.resetFiltri = function () {
        ctrl.testo = '';
        ctrl.stato = '';
        filtra();
      };
    }
  });
})();

import { Pipe, PipeTransform } from '@angular/core';

const ETICHETTE: Record<string, string> = {
  BOZZA: 'Bozza',
  PROTOCOLLATA: 'Protocollata',
  IN_ISTRUTTORIA: 'In istruttoria',
  APPROVATA: 'Approvata',
  RESPINTA: 'Respinta',
  ANNULLATA: 'Annullata',
};

/** Trasforma il codice di stato in etichetta leggibile (porting del filtro AngularJS statoLabel). */
@Pipe({ name: 'statoLabel', standalone: true })
export class StatoLabelPipe implements PipeTransform {
  transform(stato: string | null | undefined): string {
    if (!stato) {
      return '';
    }
    // `||` e non `??`: preserva esattamente il comportamento del filtro legacy
    return ETICHETTE[stato] || stato;
  }
}

import { Component, computed, inject, input } from '@angular/core';

import { Orologio } from '../../services/orologio.service';
import { ScadenzeService } from '../../services/scadenze.service';

interface VistaBadge {
  classe: 'nessuna' | 'scaduta' | 'in-scadenza' | 'ok';
  testo: string;
}

/**
 * <scadenza-badge [scadenza]="p.scadenza" />
 *
 * Comportamento preservato dalla direttiva AngularJS, incluso il quirk: "oggi" viene
 * letto solo quando cambia `scadenza` (non è un signal), esattamente come il vecchio $watch.
 */
@Component({
  selector: 'scadenza-badge',
  standalone: true,
  template: `<span class="badge" [class]="vista().classe">{{ vista().testo }}</span>`,
})
export class ScadenzaBadgeComponent {
  private readonly scadenze = inject(ScadenzeService);
  private readonly orologio = inject(Orologio);

  readonly scadenza = input<Date | null | undefined>(null);

  readonly vista = computed<VistaBadge>(() => {
    const scadenza = this.scadenza();
    if (!scadenza) {
      return { classe: 'nessuna', testo: '—' };
    }
    const giorni = this.scadenze.giorniRimanenti(scadenza, this.orologio.oggi());
    if (giorni < 0) {
      const n = Math.abs(giorni);
      return { classe: 'scaduta', testo: 'Scaduta da ' + n + (n === 1 ? ' giorno' : ' giorni') };
    }
    if (giorni <= 5) {
      return {
        classe: 'in-scadenza',
        testo: giorni === 0 ? 'Scade oggi' : 'Scade tra ' + giorni + (giorni === 1 ? ' giorno' : ' giorni'),
      };
    }
    return { classe: 'ok', testo: giorni + ' giorni' };
  });
}
