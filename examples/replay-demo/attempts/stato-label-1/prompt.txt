<source path="src/filters/stato-label.filter.js">
/* Trasforma il codice di stato in etichetta leggibile. */
(function () {
  'use strict';

  angular.module('pratiche').filter('statoLabel', function () {
    var ETICHETTE = {
      BOZZA: 'Bozza',
      PROTOCOLLATA: 'Protocollata',
      IN_ISTRUTTORIA: 'In istruttoria',
      APPROVATA: 'Approvata',
      RESPINTA: 'Respinta',
      ANNULLATA: 'Annullata'
    };

    return function (stato) {
      if (!stato) {
        return '';
      }
      return ETICHETTE[stato] || stato;
    };
  });
})();

### Test legacy (descrivono il comportamento attuale)
<source path="tests/stato-label.filter.spec.js">
describe('filtro statoLabel', () => {
  let statoLabel;

  beforeEach(() => {
    angular.mock.module('pratiche');
    angular.mock.inject(($filter) => {
      statoLabel = $filter('statoLabel');
    });
  });

  it('traduce tutti gli stati noti', () => {
    expect(statoLabel('BOZZA')).toBe('Bozza');
    expect(statoLabel('PROTOCOLLATA')).toBe('Protocollata');
    expect(statoLabel('IN_ISTRUTTORIA')).toBe('In istruttoria');
    expect(statoLabel('APPROVATA')).toBe('Approvata');
    expect(statoLabel('RESPINTA')).toBe('Respinta');
    expect(statoLabel('ANNULLATA')).toBe('Annullata');
  });

  it('restituisce stringa vuota per valori falsy', () => {
    expect(statoLabel(undefined)).toBe('');
    expect(statoLabel(null)).toBe('');
    expect(statoLabel('')).toBe('');
  });

  it('restituisce il valore originale per stati sconosciuti', () => {
    expect(statoLabel('FOO')).toBe('FOO');
  });

  it('QUIRK (bug latente, non corretto): per chiavi di Object.prototype restituisce la funzione ereditata', () => {
    // ETICHETTE[stato] || stato: 'constructor' trova Object.prototype.constructor (truthy)
    expect(typeof statoLabel('constructor')).toBe('function');
    expect(typeof statoLabel('toString')).toBe('function');
  });
});

import { Component } from '@angular/core';
import { TestBed } from '@angular/core/testing';

import { StatoLabelPipe } from './stato-label.pipe';

@Component({
  selector: 'test-host',
  standalone: true,
  imports: [StatoLabelPipe],
  template: `<span>{{ stato | statoLabel }}</span>`,
})
class HostComponent {
  stato: string | null | undefined = 'IN_ISTRUTTORIA';
}

describe('StatoLabelPipe (gold, portato dal filtro AngularJS statoLabel)', () => {
  const pipe = new StatoLabelPipe();

  it('traduce tutti gli stati noti', () => {
    expect(pipe.transform('BOZZA')).toBe('Bozza');
    expect(pipe.transform('PROTOCOLLATA')).toBe('Protocollata');
    expect(pipe.transform('IN_ISTRUTTORIA')).toBe('In istruttoria');
    expect(pipe.transform('APPROVATA')).toBe('Approvata');
    expect(pipe.transform('RESPINTA')).toBe('Respinta');
    expect(pipe.transform('ANNULLATA')).toBe('Annullata');
  });

  it('restituisce stringa vuota per valori falsy', () => {
    expect(pipe.transform(undefined)).toBe('');
    expect(pipe.transform(null)).toBe('');
    expect(pipe.transform('')).toBe('');
  });

  it('restituisce il valore originale per stati sconosciuti', () => {
    expect(pipe.transform('FOO')).toBe('FOO');
  });

  it('è utilizzabile nei template con il nome "statoLabel"', () => {
    const fixture = TestBed.createComponent(HostComponent);
    fixture.detectChanges();
    expect(fixture.nativeElement.querySelector('span').textContent).toBe('In istruttoria');
    fixture.componentInstance.stato = undefined;
    fixture.detectChanges();
    expect(fixture.nativeElement.querySelector('span').textContent).toBe('');
  });
});
